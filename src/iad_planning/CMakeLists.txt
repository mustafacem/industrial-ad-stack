cmake_minimum_required(VERSION 3.8)
project(iad_planning)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wno-unused-parameter -O3)

find_package(ament_cmake REQUIRED)
foreach(p rclcpp nav_msgs geometry_msgs std_msgs visualization_msgs autoware_planning_msgs)
  find_package(${p} REQUIRED)
endforeach()
find_package(Eigen3 REQUIRED)

add_executable(planner_node src/planner_node.cpp)
target_include_directories(planner_node PRIVATE include)
target_link_libraries(planner_node Eigen3::Eigen)
ament_target_dependencies(planner_node rclcpp nav_msgs geometry_msgs std_msgs visualization_msgs autoware_planning_msgs)
install(TARGETS planner_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_gtest(test_planning test/test_planning.cpp)
  target_include_directories(test_planning PRIVATE include)
  target_link_libraries(test_planning Eigen3::Eigen)
  add_launch_test(test/test_planner_node.py TIMEOUT 60)
endif()
ament_package()
