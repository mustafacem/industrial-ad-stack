cmake_minimum_required(VERSION 3.8)
project(iad_control)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wno-unused-parameter -O3)

find_package(ament_cmake REQUIRED)
foreach(p rclcpp nav_msgs geometry_msgs std_msgs ackermann_msgs autoware_planning_msgs autoware_vehicle_msgs)
  find_package(${p} REQUIRED)
endforeach()
find_package(Eigen3 REQUIRED)

add_executable(controller_node src/controller_node.cpp)
target_include_directories(controller_node PRIVATE include)
target_link_libraries(controller_node Eigen3::Eigen)
ament_target_dependencies(controller_node rclcpp nav_msgs geometry_msgs std_msgs ackermann_msgs autoware_planning_msgs
  autoware_vehicle_msgs)
install(TARGETS controller_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/sil_plant DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_gtest(test_control test/test_control.cpp)
  target_include_directories(test_control PRIVATE include)
  target_link_libraries(test_control Eigen3::Eigen)
  add_launch_test(test/test_controller_node.py TIMEOUT 120)
endif()
ament_package()
