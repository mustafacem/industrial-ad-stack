cmake_minimum_required(VERSION 3.8)
project(iad_calibration)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wno-unused-parameter -O3)

find_package(ament_cmake REQUIRED)
foreach(p rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros tf2_eigen)
  find_package(${p} REQUIRED)
endforeach()
find_package(Eigen3 REQUIRED)

add_executable(calibration_node src/calibration_node.cpp)
target_include_directories(calibration_node PRIVATE include)
target_link_libraries(calibration_node Eigen3::Eigen)
ament_target_dependencies(calibration_node rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2_ros tf2_eigen)
install(TARGETS calibration_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_gtest(test_calibration test/test_calibration.cpp)
  target_include_directories(test_calibration PRIVATE include)
  target_link_libraries(test_calibration Eigen3::Eigen)
  ament_target_dependencies(test_calibration sensor_msgs)
  add_launch_test(test/test_calibration_node.py TIMEOUT 120)
endif()
ament_package()
