cmake_minimum_required(VERSION 3.8)
project(iad_localization)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wno-unused-parameter -O3)

find_package(ament_cmake REQUIRED)
foreach(p rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs tf2_ros tf2_eigen)
  find_package(${p} REQUIRED)
endforeach()
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)

add_executable(localization_node src/localization_node.cpp)
target_include_directories(localization_node PRIVATE include)
target_link_libraries(localization_node Eigen3::Eigen OpenMP::OpenMP_CXX)
ament_target_dependencies(localization_node rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs visualization_msgs
  tf2_ros tf2_eigen)
install(TARGETS localization_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_gtest(test_localization test/test_localization.cpp)
  target_include_directories(test_localization PRIVATE include)
  target_link_libraries(test_localization Eigen3::Eigen OpenMP::OpenMP_CXX)
  ament_target_dependencies(test_localization sensor_msgs)
  add_launch_test(test/test_localization_node.py TIMEOUT 90)
endif()
ament_package()
