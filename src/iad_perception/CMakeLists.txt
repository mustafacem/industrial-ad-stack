cmake_minimum_required(VERSION 3.8)
project(iad_perception)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wno-unused-parameter -O3)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(autoware_perception_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(Eigen3 REQUIRED)

add_executable(perception_node src/perception_node.cpp)
target_include_directories(perception_node PRIVATE include)
target_link_libraries(perception_node Eigen3::Eigen)
ament_target_dependencies(perception_node rclcpp sensor_msgs nav_msgs std_msgs geometry_msgs visualization_msgs
  autoware_perception_msgs tf2_ros tf2_eigen)
install(TARGETS perception_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_gtest(test_perception test/test_perception.cpp)
  target_include_directories(test_perception PRIVATE include)
  target_link_libraries(test_perception Eigen3::Eigen)
  ament_target_dependencies(test_perception sensor_msgs)
  add_launch_test(test/test_perception_node.py TIMEOUT 60)
endif()
ament_package()
