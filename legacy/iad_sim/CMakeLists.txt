cmake_minimum_required(VERSION 3.8)
project(iad_sim)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wno-unused-parameter -O3)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(iad_common REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenMP REQUIRED)

add_library(iad_sim_core SHARED src/world.cpp src/scenarios.cpp src/sensors.cpp src/maps.cpp src/viz.cpp)
target_include_directories(iad_sim_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(iad_sim_core Eigen3::Eigen OpenMP::OpenMP_CXX)
ament_target_dependencies(iad_sim_core iad_common sensor_msgs visualization_msgs)
install(TARGETS iad_sim_core EXPORT export_iad_sim LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_iad_sim HAS_LIBRARY_TARGET)
ament_export_dependencies(iad_common sensor_msgs visualization_msgs Eigen3 OpenMP)

add_executable(sim_node src/sim_node.cpp)
target_link_libraries(sim_node iad_sim_core)
target_compile_options(sim_node PRIVATE -g)
target_link_options(sim_node PRIVATE -rdynamic)
target_compile_options(iad_sim_core PRIVATE -g)
ament_target_dependencies(sim_node rclcpp iad_common sensor_msgs geometry_msgs nav_msgs std_msgs std_srvs
  rosgraph_msgs visualization_msgs ackermann_msgs tf2_ros)
install(TARGETS sim_node DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_sim test/test_sim.cpp)
  target_link_libraries(test_sim iad_sim_core)
  ament_target_dependencies(test_sim iad_common)
endif()
ament_package()
